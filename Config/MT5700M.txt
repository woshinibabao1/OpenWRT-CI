# MT5700M 独立插件配置层（方案 A）
# 仅在 MT_MODE=MT5700M 时叠加（Scripts/ApplyMTMode.sh）。
#
# ── 为什么会有这个文件 ────────────────────────────────────────────────
# Scripts/ApplyMTMode.sh 的 MT5700M 分支要求本文件存在，缺了会直接
# `::error::缺少 .../Config/MT5700M.txt` + exit 1 —— 而它此前**并不存在**
# （Config/ 下只有 GENERAL / H5000M-WIFI-YES / MT5700 / PRIVATE），
# 意味着切到 MT5700M 必失败，且 Packages.sh 会先克隆 + 折叠 + cargo 编 Rust
# 后端白烧几分钟才报错。2026-09-21 补上，恢复该模式的可用性。
#
# ── 方案 A 与方案 B 的区别 ────────────────────────────────────────────
#   方案 A（本文件）：luci-app-mt5700m（LianXia233）—— LuCI 壳 + 折叠进包的
#     at-webserver Rust 后端；AT 相关能力复用 QModem feed 的
#     sms-tool_q / ubus-at-daemon。由 Packages.sh 的 FOLD_MT5700M 完成折叠。
#   方案 B（Config/MT5700.txt）：luci-app-mt5700（自研单包，src/Makefile 自带
#     cargo 交叉编译），不依赖任何 QModem 组件。
#   两者互斥，ApplyMTMode.sh 会为对方写 =n 兜底。
#
# ── 依赖核对（2026-09-21，读上游 Makefile 确认）───────────────────────
#   luci-app-mt5700m 2.4.8：LUCI_DEPENDS := +luci-base +ubus-at-daemon +sms-tool_q
#   LUCI_EXTRA_DEPENDS 里的 kmod-usb* 已由 Config/GENERAL.txt 全部置 y。
#
# ⚠️ 本层自 2026-09-15 起未再用于出固件（H5000M-MT-AUTO.yml 只跑 MT5700 组合），
#    故**未经真机验证**。启用前请先确认 Packages.sh 的 QModem 克隆与 FOLD_MT5700M
#    两段仍然有效。

# LuCI 插件（方案 A）
CONFIG_PACKAGE_luci-app-mt5700m=y
CONFIG_PACKAGE_luci-i18n-mt5700m-zh-cn=y

# QModem feed 提供的 AT 组件（luci-app-mt5700m 的硬依赖）
CONFIG_PACKAGE_sms-tool_q=y
CONFIG_PACKAGE_ubus-at-daemon=y

# 显式排除方案 B（互斥，防止 make defconfig 被其他配置层意外拉入）
CONFIG_PACKAGE_luci-app-mt5700=n
CONFIG_PACKAGE_luci-i18n-mt5700-zh-cn=n
